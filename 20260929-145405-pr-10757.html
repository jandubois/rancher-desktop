func freeLoopbackPort(t *testing.T, proto string) string {
	t.Helper()
	var addr net.Addr
	if proto == "udp" {
		c, err := net.ListenPacket("udp", "127.0.0.1:0")
		if err != nil {
			t.Fatal(err)
		}
		addr = c.LocalAddr()
		c.Close()
	} else {
		l, err := net.Listen("tcp", "127.0.0.1:0")
		if err != nil {
			t.Fatal(err)
		}
		addr = l.Addr()
		l.Close()
	}
	_, port, _ := net.SplitHostPort(addr.String())
	return port
}

func TestProbeRecheckAfterRemoveAll(t *testing.T) {
	for _, proto := range []string{"tcp", "udp"} {
		t.Run(proto, func(t *testing.T) {
			ports := gvforwarder.NewPortsForwarder(stack.New(stack.Options{}))
			mux := http.NewServeMux()
			mux.Handle("/services/forwarder/", http.StripPrefix("/services/forwarder", ports.Mux()))
			srv := httptest.NewServer(mux)
			defer srv.Close()

			wsl := &recordingWSLProxy{}
			tr := tracker.NewAPITracker(context.Background(), wsl, srv.URL, "192.168.127.2", false)

			port := freeLoopbackPort(t, proto)
			natPort, err := nat.NewPort(proto, port)
			if err != nil {
				t.Fatal(err)
			}
			pm := nat.PortMap{natPort: {{HostIP: "127.0.0.1", HostPort: port}}}
			local := "127.0.0.1:" + port
			defer ports.Unexpose(gvtypes.TransportProtocol(proto), local) //nolint:errcheck

			if err := tr.Add("scanner", pm); err != nil {
				t.Fatalf("publish Add: %v", err)
			}
			removeAllErr := tr.RemoveAll()
			reassertErr := tr.Add("scanner", pm)

			res, err := http.Get(srv.URL + "/services/forwarder/all")
			if err != nil {
				t.Fatal(err)
			}
			proxies, _ := io.ReadAll(res.Body)
			res.Body.Close()

			t.Logf("RemoveAll error:          %v", removeAllErr)
			t.Logf("reassert Add error:       %v", reassertErr)
			t.Logf("host-switch proxies:      %s", proxies)
			t.Logf("tracker record:           %v", tr.Get("scanner"))
			t.Logf("wsl-proxy messages (add/remove order): %d", len(wsl.sent))
			for i, m := range wsl.sent {
				t.Logf("  %d: Remove=%v %v", i, m.Remove, m.Ports)
			}
			last := wsl.sent[len(wsl.sent)-1]
			if reassertErr != nil || len(tr.Get("scanner")) == 0 || last.Remove {
				t.Errorf("the recheck did not restore the %s port: record %v, wsl-proxy last told Remove=%v",
					proto, tr.Get("scanner"), last.Remove)
			}
		})
	}
}

Documentation Assessment ¶

Apart from I1 and S1, every claim the rewrite makes holds up against the code. On reconnect the engine monitors expose only their own containers again (docker/events.go:74, containerd/events_linux.go:86). The iptables scanner calls Add only for entries new since its last scan. The docker loopback DNAT rule matches any destination, so traffic host-switch sends to the tap IP reaches a 127.0.0.1 publish. RemoveAll never touches the scanner's loopback listener. A delegated port never gets a forwarder, and recheckTicks resets at every expose request. All ten rewritten test doc comments describe what their tests assert.

Repo context updates ¶

  • [repo] The kube service watcher exposes every port of a Service again on every informer event, the one-hour resync included. handleUpdate puts an unchanged Service's ports back into added, and WatchForServices calls Tracker.Add for each. A tracker record the watcher lost, for example to RemoveAll, comes back on its next event, and an "already exposed" answer it gets repeats hourly for as long as the Service exists.
  • [repo] wsl-proxy (src/go/networking/cmd/proxy) listens on localhost in the WSL distro's default network namespace so that other WSL distributions reach published ports; Windows reaches them through host-switch's expose alone. Judge a missed or failed wsl-proxy notification by its effect on other WSL distributions, not on Windows.
  • [repo] Every producer of Tracker.Add passes an IPv4 host IP. The docker and containerd monitors run each engine host IP through utils.NormalizeHostIP, which keeps 127.0.0.1 and localhost and turns everything else into 0.0.0.0; main.go accepts only 0.0.0.0 or 127.0.0.1 as the kube and iptables listener IP; and the /proc/net scanner keeps only IPv4 entries. A finding that needs an IPv6 binding to reach APITracker.Add is unreachable.
  • [repo] gvisor-tap-vsock's forwarder package is in the guestagent module graph, so a Go test can run host-switch's real /expose and /unexpose handlers in-process on any host. Mount forwarder.NewPortsForwarder(stack.New(stack.Options{})).Mux() under /services/forwarder on an httptest server and point tracker.NewAPITracker at it. Use it to prove a claim about host-switch behavior instead of reasoning from upstream source.
